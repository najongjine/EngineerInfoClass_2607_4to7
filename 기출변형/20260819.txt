package itsecmorning_2607;

import java.util.ArrayList;
import java.util.Iterator;
import java.util.List;
class A {
	int a=0; // 1
	A(){
		System.out.println("A생성");
		a=1;
		paint();
	}
    public void paint() { // 오긴 옴
        System.out.print("A");
        draw();
    }
    public void draw() { // super 라고 명시하면 오버라이드 체크 안함
        System.out.print("B");
        draw(); // 그냥 메소드 호출은 override 호출한다
    }
}
class B extends A {
	B(){super();}
    public void paint() { // 오버라이드 됬네
        super.draw();
        System.out.print("C"+a);
        this.draw();
    }
    public void draw() {
        System.out.print("D"+a);
    }
}
public class itsecmorning_2607 {
	public static void main(String[] args) {
		A b = new B();
        b.paint();
        b.draw();
	}

}
