#include <stdio.h>
#define MAX_SIZE 5

int isWhat[MAX_SIZE];
int point = -1;

int isEmpty() {
    if (point == -1) return 1;
    return 0;
}

int isFull() {
    if (point == MAX_SIZE) return 1;
    return 0;
}

void into(int num) {
    if (point >= MAX_SIZE) printf("Full");
    else isWhat[++point] = num;
}

int take() {
    if (isEmpty() == 1) printf("Empty");
    else return isWhat[point--];
    return 0;
}

int main(int argc, char const* argv[]) {
    int e;
    into(1); into(2);

    while (!isEmpty()) {
        printf("%d", take());
        into(3); into(1); printf("%d", take());
    }

    return 0;
}
