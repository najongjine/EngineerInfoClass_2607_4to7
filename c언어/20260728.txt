#include <stdio.h>

typedef struct Node {
    int v;
}Node;

/* 기능 - 함수 - 모듈 - 컴포넌트 - 라이브러리 */
/* 
* 내장함수 - 언어 재작자가 이미 만들어 놓은것. 그냥 갔다 쓰면 됨
*   c언어는 내장함수가 시험에 안나와요
*   파이썬이 내장함수 위주로 나와요
*   내장함수는 내가 숙달되기 전까진 뭔놈이지 알 방법이 없어요
*   (블랙박스. 내부를 알수가 없음)
* 
* 커스텀함수 - 내가 만든거에요
*   c언어는 이거 위주로 나와요
*   (화이트박스. 내부를 다 볼수 있음)
*/
void dummy(char a) {
    a += 2;
    printf("나 %c 받았다 \n", a); // 'C'
}

int main() {
    int a = 65;
    dummy(a);
    printf("a:%c \n", a); // 'A'
    
    return 0;
}
