#include <stdio.h>

typedef struct Node {
    int v;
}Node;

/* 기능 - 함수 - 모듈 - 컴포넌트 - 라이브러리 */
/* 
* 내장함수 - 언어 재작자가 이미 만들어 놓은것. 그냥 갔다 쓰면 됨
*   c언어는 내장함수가 시험에 안나와요
*   파이썬이 내장함수 위주로 나와요
*   내장함수는 내가 숙달되기 전까진 뭔놈이지 알 방법이 없어요
*   (블랙박스. 내부를 알수가 없음)
* 
* 커스텀함수 - 내가 만든거에요
*   c언어는 이거 위주로 나와요
*   (화이트박스. 내부를 다 볼수 있음)
*/
void dummy(char a) {
    a += 2;
    printf("나 %c 받았다 \n", a); // 'C'
}

int main() {
    int a = 65;
    dummy(a);
    printf("a:%c \n", a); // 'A'
    
    return 0;
}

#include <stdio.h>

typedef struct Node {
    int v;
}Node;

/* 스탬프결합도, 자료결합도를 어떻게 구분해요?
* c언어는 구분이 쉬운 편이에요
* 기술적설명: 주소가 가면 스탬프 결합도. 그냥 데이터가 가면 자료 결합도
* 
* c언어는 숫자, 문자, 구조체 얘네만 자료 결합도. 나머지는 다 stamp 결합도
* 
* 자바,python: 숫자, bool, 문자, 문자열 얘네만 자료 결합도. 나머지는 다 stamp 결합도
* 
* primitive 자료형(기본 자료형)은 자료 결합도가 되요
* int double bool char struct
*/
void dummy(char a[]) {
    *a = *(a + 1);
    *(a + 2) = a[3];
}

int main() {
    char str1[4] = "abc";
    char* str2 = str1;
    dummy(str2);
    printf("%s \n", str1);
    
    return 0;
}
